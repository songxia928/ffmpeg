# 02.安装 ffmpeg

以下安装脚本为dockerfile格式，如果是直接在linux中安装，而不是封装镜像，去掉命令前面的 RUN ，运行命令即可。

## 1 cpu版

#### a. 编译（ubuntu）

```shell
# ==== (1) 准备
apt-get update 
apt-get install --no-install-recommends -y yasm


# ==== (2) 下载安装包
wget http://ffmpeg.org/releases/ffmpeg-4.4.4.tar.gz
tar -zxvf ffmpeg-4.4.4.tar.gz 


# ==== (3)  configure
cd ffmpeg-4.4.4

# == 编译字幕
./configure --target-os=linux --prefix=/home/ffmpeg/ffmpeg-install --arch=x86_64 --disable-doc \
    --enable-libx264 --enable-libmp3lame --enable-libopus --enable-debug=3 --extra-cflags="" --extra-ldflags="" --enable-gpl  \
    --enable-libass \
    --enable-libx265

   # 如果不需要特殊功能，则默认编辑即可： ./configure

# ==== (4) make 
make
make install

```


#### b. dockerfile
```shell
# ========== Dockerfile ==========
FROM nvidia/cudagl:11.3.0-devel-ubuntu18.04

# ==== sources
RUN sed -i "s/archive.ubuntu./mirrors.aliyun./g" /etc/apt/sources.list
RUN sed -i "s/security.ubuntu./mirrors.aliyun./g" /etc/apt/sources.list
RUN sed -i "s/deb.debian.org/mirrors.aliyun.com/g" /etc/apt/sources.list
RUN sed -i "s/security.debian.org/mirrors.aliyun.com\/debian-security/g" /etc/apt/sources.list
RUN rm /etc/apt/sources.list.d/*

# ==== base softwave
RUN apt-get update && apt-get install --no-install-recommends -y \
      build-essential \
      git \
      gcc \
      g++ \
      make \
      automake \
      autoconf \
      bzip2 \
      wget \
      curl \
      gawk \
      libtool \
      libatlas3-base \
      zlib1g-dev \
      python3 \
      python3-pip \
      python3-dev \
      subversion \
      unzip \
      flac \
      libjansson-dev \
      lrzsz  \
      vim && \
    apt-get clean all && \
    rm -rf /var/lib/apt/lists/*


# ==== install zip
RUN apt-get update && apt-get install --no-install-recommends -y \
      zip   && \
    apt-get clean all && \
    rm -rf /var/lib/apt/lists/*


# ==== install cmake, opencv(C++) 
RUN apt-get update && apt-get install --no-install-recommends -y \
      cmake    \   
      libopencv-dev    \   
      python-opencv   && \
    apt-get clean all && \
    rm -rf /var/lib/apt/lists/*


# ==== ffmpeg
# == install yasm
RUN apt-get update && apt-get install --no-install-recommends -y \
      yasm  && \
    apt-get clean all && \
    rm -rf /var/lib/apt/lists/*


# == install ffmpeg
RUN wget http://ffmpeg.org/releases/ffmpeg-3.1.3.tar.gz  && \
    tar -zxvf ffmpeg-3.1.3.tar.gz  && \
    cd ffmpeg-3.1.3      && \
    ./configure     && \
    make   && \
    make install  && \
    cd ..   && \
    rm -rf ffmpeg-3.1.3.tar.gz  && \
    rm -rf ffmpeg-3.1.3

```


## 2 gpu版

ffmpeg 在 gpu 上运行的速度较 cpu 快很多。

#### a. 编译

#### b. dockerfile
```shell
# ========== Dockerfile ==========
FROM nvidia/cuda:10.1-cudnn7-devel-ubuntu16.04

WORKDIR /build
RUN sed -i "s/archive.ubuntu./mirrors.aliyun./g" /etc/apt/sources.list
RUN sed -i "s/deb.debian.org/mirrors.aliyun.com/g" /etc/apt/sources.list
RUN sed -i "s/security.debian.org/mirrors.aliyun.com\/debian-security/g" /etc/apt/sources.list
RUN sed -E -e 's/\s+main\s?$/ main contrib non-free /' /etc/apt/sources.list >/etc/apt/sources.list.new && mv /etc/apt/sources.list.new /etc/apt/sources.list

RUN gpg --keyserver keyserver.ubuntu.com --recv A4B469963BF863CC
RUN gpg --export --armor A4B469963BF863CC | apt-key add
RUN apt-get -y update && apt-get install -y  gnupg dirmngr
RUN apt-get -y install git libass-dev fontconfig libjpeg-dev zlib1g-dev

# ==== ffmpeg
COPY  build  /build

RUN (cd nv-codec-headers-n8.2.15.12; make -j)
RUN (cd nv-codec-headers-n8.2.15.12; make install)
RUN apt-get update
RUN apt-get -y install gcc g++ make xorg-dev pkg-config\
                        libglew1.13 libglew-dev libglfw3-dev\
                        nasm yasm libx264-dev libx265-dev libvpx-dev libglu1-mesa-dev\
                        libmp3lame-dev libopus-dev libfdk-aac-dev
RUN (cd ffmpeg; ./configure --enable-libx264 --enable-libx265 --enable-libvpx --enable-libfdk-aac --enable-libmp3lame --enable-libopus --enable-nonfree --enable-gpl --enable-opengl --enable-libass --enable-libfreetype --extra-libs='-lGLEW -lglfw -ldl' --enable-cuda-nvcc --enable-libnpp --extra-cflags=-I/usr/local/cuda/include --extra-ldflags=-L/usr/local/cuda/lib64 --disable-static --enable-shared --enable-cuda --enable-cuvid)
RUN (cd ffmpeg; make -j)
RUN (cd ffmpeg; make install)
RUN apt-get install -y wget gnupg vim lrzsz python3-pip

# ==== nvidia
COPY lib64/libnvcuvid.so.440.33.01  /lib64/libnvcuvid.so.440.33.01
COPY lib64/libnvidia-encode.so.440.33.01  /lib64/libnvidia-encode.so.440.33.01
ENV LD_LIBRARY_PATH=/lib64:/usr/local/nvidia/lib:/usr/local/nvidia/lib64

# ==== fonts
COPY fonts  /usr/share/fonts
RUN apt-get install -y ttf-mscorefonts-installer
RUN apt-get install -y fontconfig
RUN mkfontscale && mkfontdir && fc-cache -fv
RUN apt-get install -y language-pack-zh-hans
RUN echo LANG="zh_CN.UTF-8" >> /etc/environment
RUN echo LANGUAGE="zh_CN:zh:en_US:en" >> /etc/environment
RUN echo en_US.UTF-8 UTF-8 >> /var/lib/locales/supported.d/local
RUN echo zh_CN.UTF-8 UTF-8 >> /var/lib/locales/supported.d/local
RUN echo zh_CN.GBK GBK >> /var/lib/locales/supported.d/local
RUN echo zh_CN GB2312 >> /var/lib/locales/supported.d/local
RUN locale-gen
RUN echo export LANG="zh_CN.UTF-8" >> /etc/profile
RUN echo "source /etc/profile" >> ~/.bashrc

# ==== conda
RUN bash /build/Anaconda3-2019.03-Linux-x86_64.sh -b -p /conda
RUN rm /build/Anaconda3-2019.03-Linux-x86_64.sh
ENV PATH=/conda/bin:/usr/local/nvidia/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
RUN conda create -n py3 python=3.6.9 pip --yes
RUN RUN ". activate py3 && python -m pip install --upgrade pip -i https://pypi.tuna.tsinghua.edu.cn/simple/ && pip --version && pip install numpy pillow requests -i https://pypi.tuna.tsinghua.edu.cn/simple/"
ENV PATH=/conda/envs/py3/bin:/conda/bin:/usr/local/nvidia/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
ENTRYPOINT ["/bin/sh", "-c", "/bin/bash"]

```
